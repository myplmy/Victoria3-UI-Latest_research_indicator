# Root = The applicable country
# scope:technology = The technology type they just acquired
on_acquired_technology = {
    on_actions = {
        uboy_ui_latest_research_indicator_on_action
    }
}

uboy_ui_latest_research_indicator_on_action = {
    # 국가 스코프에 최근 연구한 기술을 저장
    effect = {
        set_variable  = {
            name = uboy_latest_research_tech
            value = scope:technology
        }
    }
}
